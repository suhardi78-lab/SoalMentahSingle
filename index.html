📊 9. Integrasi Google Spreadsheet (Kirim Nilai Otomatis)

Masukkan URL Web App dari Google Apps Script untuk menerima hasil ujian siswa secara otomatis ke Google Spreadsheet.

✅ URL Google Apps Script berhasil disimpan di perangkat ini!
📌 PENTING: Klik tombol "💾 Download HTML Terbaru" di bawah lalu ganti file di GitHub Pages/hosting Anda agar URL otomatis terpasang di seluruh HP/perangkat siswa tanpa perlu copy-paste lagi.
💡 Panduan Pembuatan Google Apps Script:
  1. Buat Google Spreadsheet baru di Google Drive Anda.
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus kode lama dan tempel kode berikut: 
    function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); // Tunggu antrean maksimal 10 detik agar tidak ada bentrok antar pengiriman
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);

    // 1. Jika sinkronisasi otomatis dari Dashboard Guru
    if (data.type === 'dashboard_sync') {
      var sheetGuru = ss.getSheetByName("Pengaturan_Guru");
      if (!sheetGuru) {
        sheetGuru = ss.insertSheet("Pengaturan_Guru");
      }
      if (sheetGuru.getLastRow() == 0) {
        sheetGuru.appendRow(["Waktu Sync", "Judul Ujian", "Subjudul", "Token", "Durasi (Menit)", "Jumlah Soal", "Acak Soal", "Acak Opsi", "Batas Pelanggaran", "Nilai Max", "KKM", "Total Bank Soal", "Data Lengkap (JSON)"]);
      }
      sheetGuru.appendRow([
        data.waktuSync || new Date().toLocaleString(),
        data.judul || "",
        data.subJudul || "",
        data.token || "",
        data.durasi || 0,
        data.jumlahSoal || 0,
        data.acak ? "Ya" : "Tidak",
        data.acakOpsi ? "Ya" : "Tidak",
        data.maxPelanggaran || 0,
        data.nilaiMaksimal || 100,
        data.kkm || 75,
        data.totalBankSoal || 0,
        JSON.stringify(data)
      ]);
      SpreadsheetApp.flush();
      return ContentService.createTextOutput(JSON.stringify({result: "success", type: "dashboard_sync"})).setMimeType(ContentService.MimeType.JSON);
    }

    // 2. Pengiriman nilai ujian siswa
    var sheetSiswa = ss.getSheetByName("Hasil_Ujian") || ss.getActiveSheet();
    if (sheetSiswa.getLastRow() == 0) {
      sheetSiswa.appendRow(["Timestamp", "Nama", "Nomor Absen", "Kelas", "Nilai", "Benar", "Total Soal", "Pelanggaran", "Auto Submit", "Waktu Selesai"]);
    }
    sheetSiswa.appendRow([
      data.waktuSelesai || new Date(),
      data.nama || "",
      data.nomor || "",
      data.kelas || "",
      data.nilai || 0,
      data.benar || 0,
      data.totalSoal || 0,
      data.pelanggaran || 0,
      data.autoSubmit ? "Ya" : "Tidak",
      data.waktuSelesai || ""
    ]);
    SpreadsheetApp.flush();
    return ContentService.createTextOutput(JSON.stringify({result: "success", type: "hasil_ujian"})).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
  4. Klik Terapkan (Deploy) > Penerapan baru (New deployment).
  5. Pilih jenis: Aplikasi web (Web app).
  6. Setel "Siapa yang memiliki akses" (Who has access) ke: Siapa saja (Anyone).
  7. Klik Terapkan (Deploy), lalu berikan izin akses:
    • Jika muncul peringatan "Google hasn't verified this app", klik Advanced (Tingkat lanjut) di kiri bawah.
    • Klik Go to ... (unsafe) / Lanjutkan ke ... (tidak aman).
    • Klik Allow / Izinkan.
    • (Peringatan ini normal karena skrip dibuat sendiri dan belum didaftarkan sebagai aplikasi publik oleh Google).
  8. Salin URL Aplikasi Web (Web App URL) lalu tempel pada kolom di atas.
